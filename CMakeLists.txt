cmake_minimum_required(VERSION 3.16)
project(OpenGW LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL2 CONFIG QUIET)
if(NOT TARGET SDL2::SDL2)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
endif()
find_package(OpenGL REQUIRED)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(REMOVE_ITEM GAME_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/OpenGW.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/stdafx.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/controls.cpp")
add_executable(opengw ${GAME_SOURCES} platform/main_sdl.cpp platform/controls_sdl.cpp)
target_include_directories(opengw PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(opengw PRIVATE SDL2::SDL2 OpenGL::GL)
target_compile_options(opengw PRIVATE -Wno-write-strings)
if(APPLE)
    target_compile_definitions(opengw PRIVATE GL_SILENCE_DEPRECATION)
endif()
file(COPY images sounds DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
install(TARGETS opengw RUNTIME DESTINATION opengw)
install(DIRECTORY images sounds DESTINATION opengw)
install(PROGRAMS "ports/OpenGW.sh" DESTINATION .)
